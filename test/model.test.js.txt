import test from 'node:test';import assert from 'node:assert/strict';import {km,filter,score,normalize} from '../public/model.js';
const f={radius:5,min:200,max:600,rating:4,reviews:30,open:true,unknown:false,types:['japanese_restaurant']};const p={rating:4.6,reviews:500,distance:2,priceMin:300,priceMax:500,open:true,types:['japanese_restaurant']};
test('distance',()=>assert.ok(Math.abs(km({lat:0,lng:0},{lat:0,lng:1})-111.195)<.01));test('filter',()=>{assert.equal(filter(p,f),true);for(const q of [{distance:6},{rating:3.9},{reviews:20},{priceMin:700,priceMax:800},{open:false},{types:['cafe']},{priceMin:null}])assert.equal(filter({...p,...q},f),false)});test('score favors credible reviews',()=>assert.ok(score(p,f)>score({...p,rating:4.9,reviews:5},f)));test('missing price stays unknown',()=>assert.equal(normalize({id:'a'},null).priceMin,null));test('TWD range',()=>assert.equal(normalize({id:'a',priceRange:{startPrice:{currencyCode:'TWD',units:'250'},endPrice:{currencyCode:'TWD',units:'500'}}},null).priceMax,500));
